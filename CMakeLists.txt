cmake_minimum_required(VERSION 3.16)
project(ChronoVault LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(ChronoVault src/main.cpp src/ChronoVault.cpp)
target_include_directories(ChronoVault PRIVATE include third_party)
target_link_libraries(ChronoVault PRIVATE Threads::Threads)
if(WIN32)
  target_link_libraries(ChronoVault PRIVATE ws2_32)
endif()
